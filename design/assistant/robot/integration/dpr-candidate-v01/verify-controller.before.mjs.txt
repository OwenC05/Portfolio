function loadTs(path, modules, globals = {}) {
  const source = readFileSync(new URL(path, import.meta.url), 'utf8');
  const { outputText } = ts.transpileModule(source, {
    compilerOptions: { target: ts.ScriptTarget.ES2020, module: ts.ModuleKind.CommonJS, jsx: ts.JsxEmit.ReactJSX },
  });
  const exports = {};
  vm.runInNewContext(outputText, {
    exports,
    require(name) {
      assert.ok(name in modules, `Unexpected import: ${name}`);
      return modules[name];
    },
    ...globals,
  }, { filename: path });
  return exports;
}

class TrackedTarget extends EventTarget {
  listeners = new Map();
  listenerOptions = new Map();
  addEventListener(type, callback, options) {
    const handlers = this.listeners.get(type) ?? new Set();
    handlers.add(callback);
    this.listeners.set(type, handlers);
    this.listenerOptions.set(type, options);
    super.addEventListener(type, callback, options);
  }
  removeEventListener(type, callback, options) {
    this.listeners.get(type)?.delete(callback);
    super.removeEventListener(type, callback, options);
  }
  get listenerCount() { return [...this.listeners.values()].reduce((sum, handlers) => sum + handlers.size, 0); }
}

    disconnected = false;

  const disposedResources = [];

    resources.texture(track(new THREE.Texture()));

  };

    nowMs = now;

    queued.forEach((callback) => callback(now));

    // CPU fixture only: synthetic trust and propagation model the browser trace,
    // without claiming that dispatchEvent creates trusted browser input.
    Object.defineProperties(value, { isTrusted: { value: isTrusted }, target: { value: target } });
    target.dispatchEvent(value);

function mobileLayout(h, viewportWidth) {
  const gutter = viewportWidth <= 360 ? 16 : 20;
  h.window.innerWidth = viewportWidth;
  // Deliberately offset CSS fixture, not a measurement of the production shell.
  Object.assign(h.rect, { left: gutter + 1, top: 364, width: viewportWidth - 2 * gutter - 2, height: 352 });
}

function surfaceBox(h, target) {
  const positions = target.surface.geometry.getAttribute('position');
  const bounds = new THREE.Box2();
  for (let index = 0; index < positions.count; index++) {
    const point = new THREE.Vector3().fromBufferAttribute(positions, index)
      .applyMatrix4(target.surface.matrixWorld).project(h.renderers.at(-1).camera);
    bounds.expandByPoint(new THREE.Vector2(h.rect.left + (point.x + 1) * h.rect.width / 2, h.rect.top + (1 - point.y) * h.rect.height / 2));
  }
  return { center: bounds.getCenter(new THREE.Vector2()), width: bounds.max.x - bounds.min.x, height: bounds.max.y - bounds.min.y };
}

function exactTargetAt(h, point) {
  const ray = new THREE.Raycaster();
  ray.setFromCamera(new THREE.Vector2((point.x - h.rect.left) / h.rect.width * 2 - 1, 1 - (point.y - h.rect.top) / h.rect.height * 2), h.renderers.at(-1).camera);
  const hit = ray.intersectObjects(h.model.targets.map((target) => target.surface), false)[0];
  return h.model.targets.find((target) => target.surface === hit?.object)?.id ?? null;
}

function finishClick(h, point, pointerType = 'touch', overrides = {}) {
  if (pointerType === 'touch') {
    h.event('touchend', point, { pointerType, ...overrides });
    h.event('pointerout', point, { pointerType, ...overrides });
    h.event('pointerleave', point, { pointerType, ...overrides });
  }
  h.event('mousedown', point);
  h.event('mouseup', point);
  // PointerEvent click need not be primary; identity comes from the completed gesture.
  h.event('click', point, { pointerType, isPrimary: false, ...overrides });
}

function tap(h, point, pointerType = 'touch', overrides = {}) {
  const previous = h.selections.length;
  h.event('pointerdown', point, { pointerType, ...overrides });
  h.event(pointerType === 'touch' ? 'touchstart' : 'mousedown', point, { pointerType, ...overrides });
  h.event('pointerup', point, { pointerType, ...overrides });
  finishClick(h, point, pointerType, overrides);
  return h.selections.length > previous ? h.selections.at(-1) : null;
}

for (const width of [320, 390]) for (const reducedMotion of [true, false]) {
  test(`CPU event-order fixture: ${width}px ${reducedMotion ? 'reduced' : 'ordinary'} motion ends on heading after compatibility focus`, async () => {
    let active = 'BODY';
    let hash = '';
    let historyWrites = 0;
    let stage = '';
    const selectionsAt = [];
    const h = harness({ mobile: true, reducedMotion, onSelectProject(id) {
      selectionsAt.push(stage);
      const next = `#project-${id}`;
      if (hash !== next) historyWrites++;
      hash = next;
      active = `project-${id}-heading`;
    } });
    mobileLayout(h, width);
    const controller = h.mount();
    await h.flush();
    const point = h.pointFor(0);
    const trace = [];
    for (const type of ['pointerdown', 'touchstart', 'pointerup', 'touchend', 'pointerout', 'pointerleave', 'mousedown', 'mouseup', 'click']) {
      stage = type;
      h.event(type, point, { pointerType: 'touch', isPrimary: type !== 'click' });
      // Simulate the observed browser default focus action. Early selection +
      // instant scroll retargets to work; otherwise mousedown focuses canvas.
      if (type === 'mousedown') active = hash && reducedMotion ? 'work' : 'CANVAS';
      trace.push({ type, active, count: h.selections.length });
    }
    assert.equal(active, 'project-dissertation-heading', JSON.stringify(trace));
    assert.deepEqual(selectionsAt, ['click'], 'No callback before compatibility focus has completed');
    assert.deepEqual(h.selections, ['dissertation']);
    assert.equal(historyWrites, 1, 'One synthetic navigation/history write');
    h.event('click', point, { pointerType: 'touch' });
    assert.equal(h.selections.length, 1, 'Repeated click without a fresh gesture cannot select');
    assert.equal(historyWrites, 1);
    controller.dispose();
    h.assertReleased();
  });
}

test('mouse, pen and touch commit once per complete gesture; repeated/double activation needs a new gesture', async () => {
  for (const pointerType of ['mouse', 'pen', 'touch']) {
    const h = harness();
    const controller = h.mount();
    await h.flush();
    const point = h.pointFor(0);
    h.event('pointerdown', point, { pointerType });
    h.event('pointerup', point, { pointerType });
    assert.equal(h.selections.length, 0, 'Pointerup only qualifies the gesture');
    finishClick(h, point, pointerType);
    assert.deepEqual(h.selections, ['dissertation']);
    finishClick(h, point, pointerType, { detail: 2 });
    assert.equal(h.selections.length, 1, 'A duplicate click is consumed');
    assert.equal(tap(h, point, pointerType, { detail: 2 }), 'dissertation');
    assert.equal(h.selections.length, 2, 'A second complete gesture selects once');
    h.event('dblclick', point, { pointerType, detail: 2 });
    assert.equal(h.selections.length, 2);
    controller.dispose();
    h.assertReleased();
  }
});

for (const phase of ['active', 'completed']) {
  test(`${phase} gestures cannot select after scroll, resize, hiding, offscreen, cancellation, loss or disposal`, async () => {
    for (const pointerType of ['touch', 'pen']) for (const cancellation of ['scroll', 'blur', 'resize', 'host-resize', 'hidden', 'offscreen', 'pointercancel', 'outside-down', 'outside-click', 'context-loss', 'dispose']) {
      const h = harness();
      const controller = h.mount();
      await h.flush();
      const point = h.pointFor(0);
      const input = { pointerType };
      h.event('pointerdown', point, input);
      if (phase === 'completed') h.event('pointerup', point, input);
      switch (cancellation) {
        case 'scroll': h.window.dispatchEvent(new Event('scroll')); break;
        case 'blur': h.window.dispatchEvent(new Event('blur')); break;
        case 'resize': h.window.dispatchEvent(new Event('resize')); break;
        case 'host-resize': h.observers[0].callback([]); break;
        case 'hidden':
          h.document.hidden = true;
          h.document.dispatchEvent(new Event('visibilitychange'));
          h.document.hidden = false;
          h.document.dispatchEvent(new Event('visibilitychange'));
          break;
        case 'offscreen':
          h.observers[1].callback([{ target: h.host, isIntersecting: false }]);
          h.observers[1].callback([{ target: h.host, isIntersecting: true }]);
          break;
        case 'pointercancel': h.event('pointercancel', point, input); break;
        case 'outside-down': h.event('pointerdown', point, input, h.document); break;
        case 'outside-click': h.event('click', point, input, h.document); break;
        case 'context-loss': h.canvas.dispatchEvent(new Event('webglcontextlost')); break;
        case 'dispose': controller.dispose(); break;
      }
      if (phase === 'active') h.event('pointerup', point, input);
      finishClick(h, point, pointerType);
      assert.deepEqual(h.selections, [], `${pointerType} ${phase}: ${cancellation} must invalidate the sequence`);
      controller.dispose();
      h.assertReleased();
    }
  });
}

test('active leaves, drag excursions, mismatched up and second pointers cannot become completed clicks', async () => {
  for (const pointerType of ['touch', 'pen']) for (const change of ['pointerout', 'pointerleave', 'drag-return', 'up-drag', 'wrong-id', 'wrong-type', 'secondary', 'right-button', 'second-pointer', 'untrusted-down', 'untrusted-up']) {
    const h = harness();
    const controller = h.mount();
    await h.flush();
    const point = h.pointFor(0);
    const input = { pointerType };
    h.event('pointerdown', point, { ...input, isTrusted: change !== 'untrusted-down' });
    if (['pointerout', 'pointerleave'].includes(change)) h.event(change, point, input);
    if (change === 'drag-return') h.event('pointermove', { x: point.x + 8.01, y: point.y }, input);
    if (change === 'second-pointer') h.event('pointerdown', point, { ...input, pointerId: 2, isPrimary: false });
    h.event('pointerup', change === 'up-drag' ? { x: point.x + 8.01, y: point.y } : point, {
      ...input, pointerId: change === 'wrong-id' ? 2 : 1, pointerType: change === 'wrong-type' ? (pointerType === 'touch' ? 'pen' : 'touch') : pointerType,
      isPrimary: change !== 'secondary', button: change === 'right-button' ? 2 : 0, isTrusted: change !== 'untrusted-up',
    });
    finishClick(h, point, pointerType);
    assert.deepEqual(h.selections, [], `${pointerType}: ${change}`);
    controller.dispose();
    h.assertReleased();
  }
});

test('unrelated, programmatic and stale clicks consume no selection and cannot be retried', async () => {
  for (const change of ['wrong-id', 'wrong-type', 'wrong-position', 'untrusted', 'no-detail', 'no-pointer', 'right-button', 'new-down', 'new-invalid-down', 'other-up']) {
    const h = harness();
    const controller = h.mount();
    await h.flush();
    const point = h.pointFor(0);
    const touch = { pointerType: 'touch' };
    assert.equal(tap(h, point), 'dissertation', 'An otherwise valid trusted sequence must select');
    h.selections.length = 0;
    h.event('click', point, touch);
    assert.deepEqual(h.selections, [], 'Click requires a completed gesture');
    h.event('pointerdown', point, touch);
    h.event('pointerup', point, touch);
    // Satisfy the compatibility-focus prerequisite before changing one click
    // property, so identity/displacement failures reach their intended guards.
    h.event('mousedown', point);
    if (change === 'new-down') h.event('pointerdown', point, { ...touch, pointerId: 2 });
    if (change === 'new-invalid-down') h.event('pointerdown', point, { ...touch, pointerId: 2, isPrimary: false });
    if (change === 'other-up') h.event('pointerup', point, { ...touch, pointerId: 2 });
    h.event('click', change === 'wrong-position' ? { x: point.x + 9, y: point.y } : point, {
      ...touch, pointerId: change === 'wrong-id' ? 2 : change === 'no-pointer' ? undefined : 1,
      pointerType: change === 'wrong-type' ? 'mouse' : change === 'no-pointer' ? undefined : 'touch',
      isTrusted: change !== 'untrusted', detail: change === 'no-detail' ? 0 : 1, button: change === 'right-button' ? 2 : 0,
    });
    h.event('click', point, touch);
    assert.deepEqual(h.selections, [], change);
    controller.dispose();
    h.assertReleased();
  }
});

function placeSurface(h, index, x, y, width, height) {
  // Explicit CSS-pixel fixture, independent of the production bounds algorithm.
  const camera = h.renderers.at(-1).camera;
  const surface = h.model.targets[index].surface;
  const distance = 5;
  const halfHeight = Math.tan(THREE.MathUtils.degToRad(camera.fov / 2)) * distance;
  const halfWidth = halfHeight * camera.aspect;
  surface.position.set(((x - h.rect.left) / h.rect.width * 2 - 1) * halfWidth,
    (1 - (y - h.rect.top) / h.rect.height * 2) * halfHeight, -distance).applyMatrix4(camera.matrixWorld);
  surface.quaternion.copy(camera.quaternion);
  surface.scale.set(width * 2 * halfWidth / h.rect.width / 1.2, height * 2 * halfHeight / h.rect.height / 1.2, 1);
  h.model.scene.updateMatrixWorld(true);
}

test('React adapter keeps one controller and forwards the latest callbacks and state', async () => {
  // Minimal hook scheduling checks the adapter contract; this is not a React/browser StrictMode test.
  const hooks = [];
  let cursor = 0;
  let effects = [];
  const mountedControllers = [];
  const messages = [];
  const react = {
    useRef(value) {
      const slot = cursor++;
      return hooks[slot] ??= { current: value };
    },
    useLayoutEffect(effect, dependencies) {
      const slot = cursor++;
      const previous = hooks[slot];
      if (!previous || dependencies.some((value, index) => value !== previous.dependencies[index])) {
        effects.push(() => {
          previous?.cleanup?.();
          hooks[slot] = { dependencies, cleanup: effect() };
        });
      }
    },
  };
  const { default: StudioScene } = loadTs('../StudioScene.tsx', {
    react,
    'react/jsx-runtime': { jsx: (type, props) => ({ type, props }) },
    './scene/controller': {
      createStudioController(host, props) {
        const instance = { host, props, updates: [], disposals: 0, update(state) { this.updates.push(state); }, dispose() { this.disposals++; } };
        mountedControllers.push(instance);
        props.onStatusChange('loading');
        return instance;
      },
    },
  });
  const host = {};
  function commit(props) {
    cursor = 0;
    const node = StudioScene(props);
    node.props.ref.current = host;
    const queued = effects;
    effects = [];
    queued.forEach((effect) => effect());
  }
  const original = {
    selectedProjectId: null, reducedMotion: false,
    onSelectProject: (id) => messages.push(`old:${id}`),
    onStatusChange: (status) => messages.push(`old:${status}`),
  };
  commit(original);
  const sharedAssistant = robotTest.input({ count: 2 });
  commit({
    ...original, selectedProjectId: 'copilot', reducedMotion: true,
    assistant: sharedAssistant,
    onSelectProject: (id) => messages.push(`new:${id}`),
    onStatusChange: (status) => messages.push(`new:${status}`),
    onActivateAssistant: () => messages.push('new:assistant'),
    onAssistantHoverChange: (hovered) => messages.push(`new:hover:${hovered}`),
    onRobotStatusChange: (status) => messages.push(`new:robot:${status}`),
  });
  assert.equal(mountedControllers.length, 1);
  mountedControllers[0].props.onSelectProject('typeforge');
  mountedControllers[0].props.onStatusChange('ready');
  assert.deepEqual(messages, ['old:loading', 'new:typeforge', 'new:ready']);
  assert.equal(mountedControllers[0].updates.at(-1).assistant.snapshot, sharedAssistant.snapshot, 'commit-phase delivery preserves the one snapshot identity');
  mountedControllers[0].props.onActivateAssistant();
  mountedControllers[0].props.onAssistantHoverChange(true);
  mountedControllers[0].props.onRobotStatusChange('ready');
  assert.deepEqual(messages.slice(-3), ['new:assistant', 'new:hover:true', 'new:robot:ready']);
  assert.equal(mountedControllers[0].updates.at(-1).selectedProjectId, 'copilot');
  assert.equal(mountedControllers[0].updates.at(-1).reducedMotion, true);
  commit({ ...original, selectedProjectId: 'copilot', viewCommand: { action: 'right', sequence: 7 }, interactionEnabled: false });
  assert.equal(mountedControllers.length, 1, 'Command and modal state do not remount the scene');
  assert.equal(mountedControllers[0].updates.at(-1).viewCommand.sequence, 7);
  assert.equal(mountedControllers[0].updates.at(-1).interactionEnabled, false);
  commit({ ...original, selectedProjectId: 'copilot', viewCommand: { action: 'right', sequence: 7 }, interactionEnabled: true });
  assert.equal(mountedControllers[0].updates.at(-1).interactionEnabled, true);
  hooks.forEach((hook) => hook.cleanup?.());
  assert.equal(mountedControllers[0].disposals, 1);
});

function geometryFixture() {
  const camera = new THREE.PerspectiveCamera(60, 1, 0.1, 20);
  camera.updateMatrixWorld(true);
  const scene = new THREE.Scene();
  const surface = new THREE.Mesh(new THREE.PlaneGeometry(0.1, 0.1), new THREE.MeshBasicMaterial());
  surface.position.z = -5;
  scene.add(surface);
  const model = { scene, targets: [{ id: 'dissertation', surface }], occluders: [] };
  function cast(x = 250, y = 250) {
    scene.updateMatrixWorld(true); camera.updateMatrixWorld(true);
    const raycaster = new THREE.Raycaster();
    raycaster.setFromCamera(new THREE.Vector2(x / 250 - 1, 1 - y / 250), camera);
    return raycaster;
  }
  function pick(x = 250, y = 250, touch = false) { return picking.pickProject(model, camera, x, y, 500, 500, touch); }
  function blocker(width = 1, height = 1) {
    const mesh = new THREE.Mesh(new THREE.PlaneGeometry(width, height), new THREE.MeshBasicMaterial());
    mesh.position.z = -3; scene.add(mesh); model.occluders.push(mesh); return mesh;
  }
  function depths() {
    scene.updateMatrixWorld(true); camera.updateMatrixWorld(true);
    const positions = surface.geometry.getAttribute('position');
    return Array.from({ length: positions.count }, (_, i) => -new THREE.Vector3().fromBufferAttribute(positions, i).applyMatrix4(surface.matrixWorld).applyMatrix4(camera.matrixWorldInverse).z);
  }
  return { camera, scene, surface, model, cast, pick, blocker, depths };
}

test('G2 robot uses the same qualified mouse/pen/touch pipeline, hover edges and modal cancellation', async () => {
  const assistant = robotTest.input({ panelOpen: false });
  const { h, controller, asset } = await robotHarness(assistant);
  const center = asset.bindings.get('APA_head').getWorldPosition(new THREE.Vector3()).project(h.renderers[0].camera);
  const point = { x: h.rect.left + (center.x + 1) * h.rect.width / 2, y: h.rect.top + (1 - center.y) * h.rect.height / 2 };
  assert.equal(picking.pickScene(h.model, h.renderers[0].camera, point.x - h.rect.left, point.y - h.rect.top, h.rect.width, h.rect.height, false)?.kind, 'assistant');
  h.event('pointermove', point); h.event('pointermove', point); assert.deepEqual(h.assistantHovers, [true]);
  h.window.dispatchEvent(new Event('blur')); assert.deepEqual(h.assistantHovers, [true, false]);
  for (const pointerType of ['mouse', 'pen', 'touch']) {
    const n = h.assistantActivations.length; tap(h, point, pointerType); assert.equal(h.assistantActivations.length, n + 1, pointerType);
  }
  const n = h.assistantActivations.length;
  for (const invalid of ['untrusted', 'missing-down', 'drag', 'second', 'cancel', 'modal']) {
    h.event('pointerdown', point, { pointerType: 'touch', isTrusted: invalid !== 'untrusted' });
    if (invalid === 'drag') h.event('pointermove', { x: point.x + 20, y: point.y + 40 }, { pointerType: 'touch' });
    if (invalid === 'second') { h.event('pointerdown', point, { pointerId: 2, isPrimary: false, pointerType: 'touch' }); h.event('pointercancel', point, { pointerId: 2 }); }
    if (invalid === 'cancel') h.event('pointercancel', point, { pointerType: 'touch' });
    if (invalid === 'modal') await update(h, controller, { assistant: { ...assistant, panelOpen: true } });
    h.event('pointerup', point, { pointerType: 'touch' });
    if (invalid === 'missing-down') h.event('click', point, { pointerType: 'touch' }); else finishClick(h, point);
    assert.equal(h.assistantActivations.length, n, invalid);
    h.window.dispatchEvent(new Event('blur'));
  }
  assert.deepEqual(h.selections, []); controller.dispose(); h.assertReleased();
});

async function feedbackRobotHarness() {
  let queuedHover;
  const initial = robotTest.input({ panelOpen: false });
  const loaded = await robotHarness(initial, { onAssistantHoverChange: (hovered) => { queuedHover = hovered; } });
  const { h, controller, asset } = loaded;
  let assistant = initial;
  // StudioExperience commits robotHovered into attentionRequested; StudioScene
  // forwards that input in its layout effect. Deliberately allow this commit
  // between native events, without inserting a render before compatibility down.
  const commitHover = () => {
    if (queuedHover === undefined) return;
    assistant = { ...assistant, attentionRequested: queuedHover };
    queuedHover = undefined;
    controller.update({ selectedProjectId: null, reducedMotion: false, assistant });
  };
  const point = () => {
    const center = asset.bindings.get('APA_head').getWorldPosition(new THREE.Vector3()).project(h.renderers[0].camera);
    return { x: h.rect.left + (center.x + 1) * h.rect.width / 2, y: h.rect.top + (1 - center.y) * h.rect.height / 2 };
  };
  h.event('pointermove', point()); commitHover(); await h.flush(100); await h.flush(400);
  assert.equal(h.assistantHovers.at(-1), true);
  assert.equal(h.frames.size, 0, 'hover attention has settled');
  return { ...loaded, commitHover, point, get assistant() { return assistant; } };
}

for (const paintDuringHold of [false, true]) {
  test(`G2 feedback: one native mouse mousedown qualifies with root commits and paintDuringHold=${paintDuringHold}`, async () => {
    const { h, controller, commitHover, point } = await feedbackRobotHarness();
    const target = point();
    h.event('pointerdown', target);
    commitHover();
    h.event('mousedown', target); // Exactly one: the native mouse sequence.
    if (paintDuringHold) { await h.flush(410); await h.flush(710); }
    assert.equal(picking.pickScene(h.model, h.renderers[0].camera, target.x - h.rect.left, target.y - h.rect.top, h.rect.width, h.rect.height, false)?.kind, 'assistant');
    h.event('pointerup', target);
    h.event('mouseup', target);
    h.event('click', target, { isPrimary: false });
    const activated = h.assistantActivations.length;
    const hoverTrace = [...h.assistantHovers];
    controller.dispose(); h.assertReleased();
    assert.equal(activated, 1, `root hover trace ${JSON.stringify(hoverTrace)}; paintDuringHold=${paintDuringHold}`);
  });
}

test('G2 feedback: a fresh presentation still blocks gestures until its geometry is rendered', async () => {
  const loaded = await feedbackRobotHarness();
  const { h, controller, commitHover, point } = loaded;
  const target = point();
  controller.update({ selectedProjectId: null, reducedMotion: false, assistant: { ...loaded.assistant, attentionRequested: false } });
  h.event('pointerdown', target); commitHover(); h.event('mousedown', target);
  h.event('pointerup', target); h.event('mouseup', target); h.event('click', target, { isPrimary: false });
  assert.equal(h.assistantActivations.length, 0, 'pending geometry cannot qualify a new gesture');
  await h.flush(410); await h.flush(710);
  h.event('click', point(), { isPrimary: false });
  assert.equal(h.assistantActivations.length, 0, 'rendering cannot resurrect the rejected gesture');
  controller.dispose(); h.assertReleased();
});
